# build7-fixups.txt -- config additions for Lock_Down build #7.
#
# Usage (from the kernel tree):
#   cp ../config/hardened.config .config
#   cat ../config/build7-fixups.txt >> .config
#   make olddefconfig
#   ../scripts/preflight-check.sh .config   # must pass before compiling
#
# This file repeats every [required] symbol from config/required-symbols.txt.
# Appending duplicates of lines already in hardened.config is harmless:
# Kconfig takes the last value seen, and olddefconfig then resolves
# dependencies. The preflight script is the authority, not this file.

# Bus (22 Sept: DRM_I915 was silently dropped by olddefconfig because PCI
# was unmet; no PCI also means no NVMe, so this is boot-critical)
CONFIG_PCI=y

# Boot console
CONFIG_FB=y
CONFIG_FB_EFI=y
CONFIG_FRAMEBUFFER_CONSOLE=y
CONFIG_DRM=y
CONFIG_DRM_I915=y
CONFIG_TTY=y

CONFIG_UNIX98_PTYS=y

# Initramfs
CONFIG_BLK_DEV_INITRD=y
CONFIG_RD_GZIP=y
CONFIG_RD_ZSTD=y

# LUKS root (CONFIG_MD=y is load-bearing: BLK_DEV_DM is under `if MD`)

CONFIG_CRYPTO_XTS=y
